PASSO 2 - Criar o módulo de sessão
==================================================================
[ ] Passo 2 - Criar database/sessao.py
    ------------------------------------------------
    Criar o arquivo database/sessao.py com 3 funções:

        import keyring

        SERVICO = "consultorio"

        def salvar_sessao(usuario_id):
            keyring.set_password(SERVICO, "usuario_id", str(usuario_id))

        def carregar_sessao():
            try:
                valor = keyring.get_password(SERVICO, "usuario_id")
            except Exception:
                return None          # backend indisponivel -> nao quebra
            return int(valor) if valor else None

        def limpar_sessao():
            try:
                keyring.delete_password(SERVICO, "usuario_id")
            except Exception:
                pass                  # ja limpo -> nao faz nada

    Dicas:
    - carregar_sessao() retorna int (id) ou None.
    - NUNCA deixe uma excecao do keyring derrubar o app.
    - Nao guarda senha/token, so o usuario_id.

              janela_login = None   # global: pode nao existir

              def abrir_app(usuario):
                  definir_usuario(usuario)
                  if janela_login is not None:
                      janela_login.destroy()
                  app = App()
                  app.mainloop()
                  # se o App voltou aqui com a flag de logout:
                  if getattr(app, "voltou_para_login", False):
                      limpar_sessao()

              def iniciar():
                  global janela_login
                  # 1) tenta restaurar sessao salva no Credential Manager
                  uid = carregar_sessao()
                  if uid is not None:
                      usuario = get_user_by_id(uid)
                      if usuario:
                          definir_usuario(usuario)
                          app = App()
                          app.mainloop()
                          if getattr(app, "voltou_para_login", False):
                              limpar_sessao()
                              iniciar()
                          return
                      else:
                          limpar_sessao()   # usuario nao existe mais

                  # 2) sem sessao valida -> tela de login
                  janela_login = mostrar_login(on_success=abrir_app)
                  janela_login.mainloop()

              if __name__ == "__main__":
                  iniciar()

PASSO 5 - Testes
==================================================================
[ ] Passo 5 - Criar tests/test_sessao.py
    ------------------------------------------------
    Testar a sessao com o keyring trocado por um backend fake em
    memoria (nao usa o Windows de verdade nos testes).

    Exemplo (usando unittest.mock):

        from unittest.mock import patch
        from database import sessao

        def test_salvar_e_carregar():
            with patch.object(sessao.keyring, "get_password", return_value="7"):
                assert sessao.carregar_sessao() == 7

        def test_carregar_sem_sessao():
            with patch.object(sessao.keyring, "get_password", return_value=None):
                assert sessao.carregar_sessao() is None

        def test_carregar_erro_backend_retorna_none():
            with patch.object(sessao.keyring, "get_password", side_effect=Exception("x")):
                assert sessao.carregar_sessao() is None

        def test_limpar_nao_deu_erro():
            with patch.object(sessao.keyring, "delete_password", side_effect=Exception("x")):
                sessao.limpar_sessao()   # nao deve levantar excecao

    Depois rodar os testes:
        .venv\Scripts\python -m pytest -q
        .venv\Scripts\python -m ruff check .
